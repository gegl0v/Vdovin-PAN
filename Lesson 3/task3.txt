#include <iostream>
#include <vector>
#include <iomanip>

using namespace std;

int main() {
    double v0 = 20.0;
    double g = 9.81;
    double dt = 1.0;
    double t_max = 5.0;

    vector<double> times;
    vector<double> velocities;


    for (double t = 0.0; t <= t_max; t += dt) {
        double v = v0 - g * t;

        if (v < -1.0) break;

        times.push_back(t);
        velocities.push_back(v);
    }

    cout << fixed << setprecision(2);
    cout << setw(10) << "t (s)" << setw(15) << "v (m/s)" << endl;

    for (size_t i = 0; i < times.size(); ++i) {
        cout << setw(10) << times[i]
             << " | "
             << setw(15) << velocities[i] << endl;
    }

    double t_zero = v0 / g;
    cout << "\nТочное время, когда v = 0: " << t_zero << " с" << endl;

    return 0;
}
